CREATE VIEW viewA AS
SELECT * FROM Employees
WHERE Ename LIKE 'S%';

SELECT* FROM viewA;

CREATE VIEW viewB AS SELECT
SELECT*FROM Employees
WHERE DeptNo != 20;

SELECT* FROM viewB;


CREATE VIEW viewC AS 
SELECT * FROM Employees
WHERE Salary BETWEEN 40000 AND 60000;

SELECT* FROM viewC;


CREATE VIEW viewD AS 
SELECT  SUM(Salary), COUNT(*), MAX(Salary) FROM Employees
WHERE Job = 'Manager';

SELECT* FROM viewD;

CREATE VIEW viewE AS 
SELECT COLUMN ('Salary' ,'DeptNo')
FROM Employees;


SELECT* FROM viewE;


CREATE VIEW viewF AS 
SELECT Job, Salary FROM Employees
ORDER BY Salary DESC;


SELECT* FROM viewF;


CREATE VIEW viewG AS 
SELECT * FROM Employees
HAVING  Salary>40000;

SELECT* FROM viewG;
